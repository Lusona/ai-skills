import subprocess
import sys
import unittest

import greet


class GreetTest(unittest.TestCase):
    def test_greet(self):
        self.assertEqual(greet.greet("Ada"), "Hello, Ada!")

    def test_cli(self):
        result = subprocess.run([sys.executable, "greet.py", "Ada"], capture_output=True, text=True)
        self.assertEqual(result.returncode, 0)
        self.assertEqual(result.stdout, "Hello, Ada!\n")
        self.assertEqual(result.stderr, "")

    def test_cli_rejects_empty_name(self):
        result = subprocess.run([sys.executable, "greet.py", ""], capture_output=True, text=True)
        self.assertEqual(result.returncode, 2)
        self.assertEqual(result.stdout, "")
        self.assertEqual(result.stderr, "error: name must not be empty\n")

    def test_cli_wrong_argument_count_shows_usage(self):
        result = subprocess.run([sys.executable, "greet.py"], capture_output=True, text=True)
        self.assertEqual(result.returncode, 1)
        self.assertEqual(result.stdout, "")
        self.assertEqual(result.stderr, "usage: greet.py NAME\n")


if __name__ == "__main__":
    unittest.main()
